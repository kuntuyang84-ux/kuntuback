Day 8 · 数字反转

日期：2026-09-09 · C语言

代码

#include <stdio.h>
 
int main()
{
    int a,ge,shi,bai,result; 
    
    printf("输入一个三位数：");
    scanf("%d",&a);
    
    ge=a%10;
    shi=(a/10)%10;
    bai=a/100;
    
    result=ge*100+shi*10+bai;
    printf("反转后是：%d",result);
    return 0;
    
}
